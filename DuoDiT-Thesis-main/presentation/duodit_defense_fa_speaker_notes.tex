\begin{frame}[t,shrink=4]{\sref{۱}{عنوان و معرفی}}
\begin{noteslist}
\goal{فضای رسمی دفاع را شروع کنید و در همان ابتدا موضوع، نقش خودتان، استاد راهنما و چارچوب مسئله را روشن کنید.}
\say{با سلام و احترام خدمت اساتید محترم و حاضران گرامی. من مصطفی شهبازی دیل هستم و در این ارائه پایان‌نامه خود با عنوان «بهبود مدل‌های انتشار مبتنی بر مبدل‌های بینایی برای تولید تصویر» را ارائه می‌کنم. تمرکز کار روی خانواده مدل‌های \termfn{\dit{}}{۱} است؛ یعنی مدل‌هایی که فرآیند حذف نویز در مدل انتشار را به‌جای \termfn{\lr{U-Net}}{۲} با مبدل‌های بینایی انجام می‌دهند. ایده اصلی این پایان‌نامه این است که به جای آموزش مجدد کل یک مدل بزرگ، یک مسیر سبک و محلی در کنار بدنه اصلی اضافه کنیم.}
\defense{از همان ابتدا کلمات کلیدی را آرام و شمرده بگویید: «مدل انتشار»، «مبدل بینایی»، «تنظیم دقیق پارامتر-کارا»، «شاخه دوم». داور باید بداند کار هم معماری است و هم کارایی آموزش.}
\transition{بعد از معرفی، بگویید ابتدا نقشه ارائه را مرور می‌کنم تا مسیر بحث مشخص باشد.}
\end{noteslist}
\notedefs{\footdef{۱}{مدل انتشار مبتنی بر مبدل \lr{(Diffusion Transformer)}؛ شبکه حذف نویز آن با بلوک‌های مبدل ساخته می‌شود.}
\footdef{۲}{معماری رمزگذار-رمزگشا با اتصالات پرشی \lr{(U-Net)} برای حفظ اطلاعات مکانی.}}
\end{frame}

\begin{frame}[t,shrink=4]{\sref{۲}{نقشه ارائه}}
\begin{noteslist}
\goal{به داور نشان دهید ارائه ساختار روشن دارد و از مقدمه به روش و سپس نتایج می‌رسد.}
\say{ارائه در پنج بخش تنظیم شده است. در بخش اول، مسئله تولید تصویر و مسیر تاریخی رسیدن از مدل‌های کلاسیک به مدل‌های انتشار و \dit{} را کوتاه مرور می‌کنم. سپس کارهای مرتبط را توضیح می‌دهم؛ این بخش برای نشان دادن شکاف پژوهشی مهم است. بعد معماری پیشنهادی \termfn{\duo{}}{۱} را مرحله به مرحله معرفی می‌کنم. در بخش نتایج، تنظیمات آزمایش، نتایج کمی، کیفی، مطالعه انسانی، هزینه محاسباتی و مطالعات حذفی را گزارش می‌کنم. در پایان هم محدودیت‌ها، پیشنهادهای آینده و جمع‌بندی را مطرح می‌کنم.}
\defense{اینجا وارد جزئیات علمی نشوید؛ هدف فقط ایجاد انتظار درست است. به‌خصوص تأکید کنید که مهم‌ترین بخش‌های دفاع، «روش» و «نتایج» هستند.}
\transition{حالا از زمینه کلی تولید تصویر شروع می‌کنیم؛ یعنی اینکه اصلاً چرا تولید تصویر به یک مسئله یادگیری توزیع تبدیل شده است.}
\end{noteslist}
\notedefs{\footdef{۱}{معماری پیشنهادی این پایان‌نامه \lr{(DuoDiT)}؛ یک \lr{DiT} منجمد به‌همراه شاخه دوم محلی و آموزش‌پذیر.}}
\end{frame}

\begin{frame}[t,shrink=4]{\sref{۳}{پیش از یادگیری ماشین: تولید با قواعد}}
\begin{noteslist}
\goal{شروع تاریخچه را قبل از یادگیری ماشین قرار دهید و نشان دهید تولید تصویر ابتدا قانون‌محور بوده است.}
\say{در ابتدا تولید تصویر بیشتر به معنی ساخت تصویر با قواعد صریح بود. طراح یا برنامه‌نویس باید خودش تعیین می‌کرد که خطوط، شکل‌ها، رنگ‌ها، بافت‌ها و روابط هندسی چگونه کنار هم قرار بگیرند. در گرافیک کامپیوتری و مدل‌سازی رویه‌ای، می‌شد صحنه‌های کنترل‌شده یا بافت‌های مشخص ساخت. در پردازش تصویر کلاسیک هم از فیلترها، الگوها و کپی‌برداری از بافت برای ساخت یا کامل‌کردن بخش‌هایی از تصویر استفاده می‌شد. مزیت این رویکردها کنترل بالا روی موارد ساده بود، اما ایراد اصلی آن‌ها این بود که تنوع تصاویر طبیعی، ابهام معنایی و جزئیات پیچیده را نمی‌توانستند به خوبی پوشش دهند.}
\defense{تأکید کنید که قبل از یادگیری ماشین، سیستم واقعاً «یاد نمی‌گرفت»؛ بیشتر اجرای مجموعه‌ای از قواعد طراحی‌شده توسط انسان بود. همین وابستگی به طراحی دستی، انگیزه حرکت به سمت روش‌های داده‌محور شد.}
\transition{گام بعدی این بود که بخشی از قواعد به جای طراحی دستی، از داده تخمین زده شود.}
\end{noteslist}
\end{frame}

\begin{frame}[t,shrink=4]{\sref{۴}{روش‌های آماری و یادگیری ماشین اولیه}}
\begin{noteslist}
\goal{نشان دهید روش‌های آماری و یادگیری ماشین اولیه داده‌محورتر شدند، اما هنوز محدودیت جدی داشتند.}
\say{با ورود روش‌های آماری و یادگیری ماشین، تولید تصویر از قواعد کاملاً دستی فاصله گرفت. در این مرحله، سیستم‌ها از داده الگو می‌گرفتند؛ مثلاً ویژگی‌های دستی، مدل‌های بافت، مدل‌های چهره یا روش‌های بازسازی استفاده می‌شدند. خودرمزگذارها تصویر را فشرده می‌کردند و دوباره بازسازی می‌کردند. بعدها خانواده \termfn{\lr{VAE}}{۱} فضای نهان منظم‌تری برای نمونه‌سازی فراهم کرد. اما این روش‌ها معمولاً خروجی نرم و کم‌جزئیات داشتند، در تصویرهای پیچیده خوب تعمیم نمی‌دادند، و برای تولید تصویر طبیعی با کیفیت بالا کافی نبودند.}
\defense{اینجا ضعف را مشخص بگویید: یادگیری ماشین اولیه توانست بخشی از قواعد را از داده استخراج کند، اما کیفیت بصری و تنوع خروجی همچنان محدود بود.}
\transition{برای خروج از این محدودیت‌ها، یادگیری عمیق مولد وارد شد؛ ابتدا با \lr{GAN} و سپس با مدل‌های انتشار.}
\end{noteslist}
\notedefs{\footdef{۱}{خودرمزگذار متغیرپایه \lr{(Variational Autoencoder)}؛ مدل احتمالاتی برای یادگیری فضای نهان منظم.}}
\end{frame}

\begin{frame}[t,shrink=4]{\sref{۵}{یادگیری عمیق مولد تا مدل‌های انتشار}}
\begin{noteslist}
\goal{مسیر را از \lr{GAN} به diffusion و سپس \dit{} جمع‌بندی کنید و ضعف هر مرحله را بگویید.}
\say{\termfn{\lr{GAN}}{۱}ها تولید تصویر را به رقابت بین مولد و متمایزگر تبدیل کردند. نقطه قوت آن‌ها ساخت تصاویر شارپ‌تر بود، اما آموزش آن‌ها حساس و ناپایدار است و گاهی مدل فقط بخشی از تنوع داده را پوشش می‌دهد. مدل‌های انتشار رویکرد متفاوتی دارند: تولید تصویر را به حذف نویز تدریجی تبدیل می‌کنند. این کار معمولاً آموزش پایدارتر و کیفیت بهتری می‌دهد، اما هزینه نمونه‌برداری و آموزش بالاتر است. در ادامه، \termfn{\dit{}}{۲} شبکه حذف نویز را با مبدل پیاده می‌کند. این انتخاب برای مقیاس‌پذیری جذاب است، اما تنظیم دقیق کامل چنین مدل بزرگی پرهزینه می‌شود.}
\defense{جمع‌بندی دفاعی این اسلاید این است که هر مرحله یک مشکل مرحله قبل را کم کرده، اما مشکل تازه‌ای ساخته است: قواعد دستی تنوع ندارند؛ روش‌های آماری کیفیت محدود دارند؛ \lr{GAN} ناپایدار است؛ diffusion پرهزینه است؛ و \dit{} از نظر تنظیم دقیق سنگین است.}
\transition{بعد از این مرور کوتاه تاریخی، وارد کارهای مرتبط می‌شویم تا جایگاه پایان‌نامه نسبت به روش‌های انتشار و تنظیم پارامتر-کارا روشن شود.}
\end{noteslist}
\notedefs{\footdef{۱}{شبکه مولد تخاصمی \lr{(Generative Adversarial Network)}؛ مدل مولد مبتنی بر رقابت مولد و متمایزگر.}
\footdef{۲}{مدل انتشار مبتنی بر مبدل \lr{(Diffusion Transformer)}؛ در آن شبکه حذف نویز با بلوک‌های مبدل ساخته می‌شود.}}
\end{frame}

\begin{frame}[t,shrink=4]{\sref{۶}{مجموعه‌داده \imagenet{}}}
\begin{noteslist}
\goal{دلیل انتخاب \imagenet{} و دشواری ارزیابی روی آن را روشن کنید.}
\say{\imagenet{} در این ارائه نقش مجموعه‌داده مرجع را دارد. اهمیت آن این است که فقط چند تصویر ساده یا یک دامنه محدود نیست؛ هزار کلاس شیء با تنوع زیاد در ظاهر، زمینه، مقیاس و زاویه دید دارد. بنابراین مدل تولید تصویر باید هم مفهوم کلاس را بفهمد و هم جزئیات بصری قابل قبول بسازد. در این پژوهش تولید کلاس‌محور با وضوح \lr{256x256} بررسی شده است. نکته دفاعی این است که \imagenet{} فشار خوبی روی توان مدل برای حفظ ساختار معنایی و جزئیات محلی ایجاد می‌کند.}
\defense{روی این نکته مکث کنید که انتخاب \imagenet{} فقط انتخاب یک دیتاست معروف نیست؛ معیار سختی برای تولید کلاس‌محور و سنجش تنوع است.}
\transition{بعد از معرفی داده، باید بگوییم کیفیت تولید با چه معیارهایی سنجیده شده است.}
\end{noteslist}
\end{frame}

\begin{frame}[t,shrink=4]{\sref{۷}{ارزیابی مدل‌های تولید تصویر}}
\begin{noteslist}
\goal{سه معیار ارزیابی را بدون ورود به فرمول توضیح دهید.}
\say{برای ارزیابی خودکار، سه معیار اصلی استفاده شده است. \fid{} و \kid{} معیارهای توزیعی هستند؛ یعنی مجموعه تصاویر تولیدی را در برابر مجموعه تصاویر واقعی می‌سنجند و به یک تصویر تنها محدود نیستند. \fid{} به فاصله کلی توزیع ویژگی‌ها نگاه می‌کند و \kid{} با نگاه کرنلی، مقایسه توزیعی مکملی ارائه می‌دهد. \lpips{} بیشتر به فاصله ادراکی نزدیک است و از ویژگی‌های شبکه عمیق برای سنجش شباهت استفاده می‌کند. در هر سه معیار، مقدار کمتر بهتر است. با این حال، هیچ معیار واحدی کل کیفیت تصویر را توضیح نمی‌دهد؛ به همین دلیل در ادامه نتایج کمی را کنار نمونه‌های تصویری و مطالعه انسانی تفسیر می‌کنیم.}
\defense{اگر داور درباره فرمول پرسید، بگویید در این بخش فقط شهود معیارها مطرح شده و فرمول‌ها در صورت نیاز قابل توضیح‌اند؛ اما برای جریان دفاع، تفسیر کمتر-بهتر و مکمل‌بودن معیارها کافی است.}
\transition{حالا که داده و معیار مشخص شد، وارد کارهای مرتبط می‌شویم.}
\end{noteslist}
\end{frame}

\begin{frame}[t,shrink=4]{\sref{۸}{نقشه تاریخی کارهای مرتبط}}
\begin{noteslist}
\goal{با نمودار درختی نشان دهید \duo{} از دو ریشه می‌آید: تولید تصویر و انتقال/تنظیم دانش.}
\say{این نمودار جایگاه پایان‌نامه را نشان می‌دهد. سمت چپ مسیر تولید تصویر است: از \lr{VAE} و \lr{GAN} به مدل‌های انتشار و سپس معماری‌های مبدلی مثل \lr{GenViT}، \termfn{\lr{U-ViT}}{۱} و \dit{}. سمت راست مسیر انتقال یادگیری و تنظیم دقیق است: از \lr{Fine-Tuning} کامل به روش‌های \termfn{\lr{PEFT}}{۲}، \lr{Side-Tuning}، \lr{ControlNet} و \lr{FineDiffusion}. \duo{} در نقطه اتصال این دو مسیر قرار می‌گیرد؛ یعنی می‌خواهد یک \dit{} منجمد را با شاخه‌ای سبک و قابل آموزش تقویت کند.}

\say{مدل پایه \lr{DiT-XL/2} است؛ یعنی ۲۸ بلوک مبدل، بعد پنهان ۱۱۵۲ و ۱۶ سر توجه. در این پایان‌نامه وزن‌های بدنه اصلی ثابت نگه داشته می‌شوند. این تصمیم دو دلیل دارد: اول، دانش پیش‌آموزش‌دیده مدل پایه حفظ می‌شود و خطر خراب شدن آن در تنظیم دقیق کمتر است. دوم، هزینه آموزش کمتر می‌شود، چون برای بخش بزرگی از مدل نیازی به ذخیره گرادیان و به‌روزرسانی وزن‌ها نداریم. در فرمول، \(\theta\)\fnm{۱} پارامترهای ثابت بدنه است و خروجی آخرین بلوک با \(H_{\text{backbone}}^{(L)}\)\fnm{۲} نمایش داده می‌شود.}
\defense{کلمه «منجمد» را دقیق توضیح دهید: یعنی وزن‌ها در forward استفاده می‌شوند، اما در backward به‌روزرسانی نمی‌شوند.}
\transition{اما چون بدنه اصلی ثابت است، باید مسیری آموزش‌پذیر برای سازگاری مدل اضافه کنیم؛ این همان شاخه \lr{x2} است.}
\end{noteslist}
\notedefs{\footdef{۱}{نماد \(\theta\): پارامترهای بدنه اصلی \lr{DiT} که در این روش منجمد می‌مانند.}
\footdef{۲}{نماد \(H_{\text{backbone}}^{(L)}\): خروجی بدنه اصلی پس از بلوک آخر؛ \(L\) تعداد بلوک‌های بدنه است.}}
\end{frame}

\begin{frame}[t,shrink=4]{\sref{۲۱}{شاخه \lr{x2}: افزایش وضوح توکنی}}
\begin{noteslist}
\goal{انگیزه و مسئله فنی شاخه دوم را توضیح دهید.}
\say{در شاخه اصلی \dit{}، پچ‌ها اندازه ۲ دارند. شاخه \lr{x2} با اندازه پچ ۱ کار می‌کند؛ یعنی در همان فضای نهان، تعداد توکن‌ها چهار برابر می‌شود. مزیت این کار این است که شاخه دوم جزئیات مکانی ریزتری می‌بیند. اما مشکل این است که بدنه اصلی \(T\) توکن دارد و شاخه دوم \(4T\) توکن. اگر این توکن‌ها را مستقیم به بدنه اضافه کنیم، طول توالی و هزینه توجه زیاد می‌شود و ادغام هم ساده نیست. بنابراین به یک مکانیزم کاهش طول توالی نیاز داریم که اطلاعات محلی را از دست ندهد.}
\defense{تأکید کنید که مسئله فقط «اضافه کردن شاخه» نیست؛ مسئله اصلی پردازش و هم‌ترازسازی توکن‌های وضوح‌بالاست.}
\transition{برای پردازش این توکن‌های وضوح‌بالا، از یک خبره \vit{} استفاده می‌کنیم.}
\end{noteslist}
\end{frame}

\begin{frame}[t,shrink=4]{\sref{۲۲}{\vit{} به‌عنوان خبره شاخه دوم}}
\begin{noteslist}
\goal{دلیل استفاده از بلوک \vit{} پیش‌آموزش‌دیده را توضیح دهید.}

\defense{نکته کلیدی این است که \vit{} اینجا مدل اصلی تولید نیست؛ یک خبره کمکی برای شاخه دوم است. این تمایز را روشن بگویید.}
\transition{حالا مکانیزم توکن کلاس محلی را که خروجی شاخه را هم‌اندازه می‌کند جداگانه می‌بینیم.}
\end{noteslist}
\notedefs{\footdef{۱}{نماد \(\mathcal{T}_\phi\): بلوک یا خبره آموزش‌پذیر شاخه دوم با پارامترهای \(\phi\).}}
\end{frame}

\begin{frame}[t,shrink=4]{\sref{۲۳}{توکن کلاس محلی}}
\begin{noteslist}
\goal{مکانیزم اصلی نوآوری را ساده و دقیق توضیح دهید.}
\say{برای هر توکن در شاخه اصلی، چهار توکن متناظر در شاخه \lr{x2} وجود دارد. در روش پیشنهادی، این چهار توکن در یک گروه قرار می‌گیرند و یک توکن کلاس محلی قابل یادگیری به آن‌ها اضافه می‌شود. سپس این گروه پنج‌توکنی از بلوک پردازشی عبور می‌کند و در پایان فقط خروجی توکن کلاس نگه داشته می‌شود. به این ترتیب، هر گروه چهارتایی به یک توکن خلاصه می‌شود. تفاوت این روش با \lr{Average Pooling} یا \lr{Max Pooling} این است که خلاصه‌سازی ثابت نیست؛ توکن کلاس با توجه به محتوای همان گروه یاد می‌گیرد چه اطلاعاتی را نگه دارد.}
\defense{اگر داور پرسید چرا اسم آن «کلاس» است، بگویید از ایده توکن کلاس در \vit{} الهام گرفته شده، اما اینجا نقش آن خلاصه‌سازی محلی است، نه طبقه‌بندی کل تصویر.}
\transition{بعد از به دست آوردن خروجی هم‌اندازه با بدنه اصلی، باید آن را با جریان اصلی ادغام کنیم.}
\end{noteslist}
\end{frame}

\begin{frame}[t,shrink=4]{\sref{۲۴}{ادغام نهایی}}
\begin{noteslist}
\goal{محل و دلیل fusion را توضیح دهید.}
\say{بعد از انتخاب توکن‌های کلاس محلی، خروجی شاخه دوم از نظر طول توالی با خروجی بدنه اصلی هم‌اندازه است. در نسخه نهایی \duo{}، ادغام پس از آخرین بلوک بدنه اصلی انجام می‌شود. این ادغام به صورت جمع باقی‌مانده است: \(H_{\text{fused}}=H_{\text{backbone}}^{(L)}+H_{x2}\)\fnm{۱}. سپس لایه نهایی \dit{} از ویژگی ادغام‌شده برای پیش‌بینی نویز استفاده می‌کند. ادغام نهایی ساده‌تر و کم‌هزینه‌تر از ادغام در همه لایه‌هاست و اجازه می‌دهد شاخه دوم مثل یک تصحیح محلی روی خروجی بدنه اصلی عمل کند.}
\defense{اگر پرسیده شد چرا ادغام چندمرحله‌ای انجام نشده، بگویید بخشی از فضای طراحی است، اما در نسخه گزارش‌شده ادغام نهایی تعادل خوبی بین هزینه و ظرفیت ایجاد کرده است.}
\transition{در نهایت باید ببینیم تابع هزینه و تعداد پارامترهای آموزش‌پذیر چگونه تعریف می‌شوند.}
\end{noteslist}
\notedefs{\footdef{۱}{نماد \(H_{\text{fused}}\): ویژگی ادغام‌شده؛ \(H_{x2}\) خروجی هم‌اندازه‌شده شاخه وضوح‌بالا است.}}
\end{frame}

\begin{noteslist}
\goal{نشان دهید پارامتر-کارایی فقط یک شعار نیست و با هزینه آموزش هم مرتبط است.}
\say{در این جدول \duo{} با \lr{FineDiffusion} مقایسه شده است. \duo{} تعداد پارامتر آموزش‌پذیر بیشتری دارد: ۱۷٫۹۵ میلیون در برابر ۱۲٫۱۵ میلیون. اما در پروفایل گزارش‌شده، زمان هر دوره برای \duo{} برابر ۱۳ دقیقه و برای \lr{FineDiffusion} برابر ۲۲ دقیقه است. مصرف حافظه اوج هم برای \duo{} برابر ۲٫۹۱ گیگابایت و برای \lr{FineDiffusion} برابر ۳٫۱۹ گیگابایت گزارش شده است. بنابراین با وجود افزایش محدود پارامترهای آموزش‌پذیر، هزینه عملی آموزش کنترل‌شده باقی مانده است.}
\defense{اگر درباره تفاوت \termfn{\lr{GFLOPs}}{۱} سؤال شد، بگویید \duo{} \lr{GFLOPs} بیشتری دارد، اما زمان و حافظه پروفایل‌شده در تنظیم گزارش‌شده کمتر بوده است؛ بنابراین هر دو نوع عدد باید کنار هم دیده شوند.}
\transition{برای فهم اینکه کدام جزء معماری اثرگذار بوده، مطالعات حذفی را می‌بینیم.}
\end{noteslist}
\notedefs{\footdef{۱}{میلیارد عملیات ممیز شناور \lr{(Giga Floating-Point Operations)}؛ شاخصی برای هزینه محاسباتی نظری.}}
\end{frame}

\begin{frame}[t,shrink=4]{\sref{۳۳}{مطالعه حذفی: راهبرد تجمیع}}
\begin{noteslist}
\goal{نقش توکن کلاس محلی را با مقایسه با pooling و projection ثابت نشان دهید.}
\say{در این مطالعه، چند روش برای کاهش طول توالی شاخه دوم مقایسه شده‌اند. \lr{Average Pooling} و \lr{Max Pooling} قواعد ثابت دارند؛ یعنی مستقل از محتوای تصویر، چهار توکن را با یک قانون خلاصه می‌کنند. \lr{Linear Projection} قابل یادگیری است، اما همچنان ترکیب مستقیم و فشرده‌سازی خطی انجام می‌دهد. در مقابل، \termfn{\lr{Local CLS Token}}{۱} اجازه می‌دهد یک توکن یادگیرنده با توکن‌های محلی تعامل کند و نماینده همان ناحیه شود. نتیجه جدول نشان می‌دهد این روش در هر سه معیار بهترین مقدار را دارد: \fid{}، \kid{} و \lpips{}.}
\defense{این اسلاید یکی از مهم‌ترین شواهد برای نوآوری روش است. حتماً بگویید نتیجه با انگیزه معماری سازگار است: تجمیع محتوا-آگاه بهتر از تجمیع ثابت است.}
\transition{بعد از راهبرد تجمیع، اثر عمق شاخه دوم و مقدار \cfg{} بررسی می‌شود.}
\end{noteslist}
\notedefs{\footdef{۱}{عبارت \lr{CLS} کوتاه‌شده \lr{classification} است؛ اینجا توکن کلاس محلی نقش خلاصه‌ساز محتوای یک گروه توکنی را دارد.}}
\end{frame}

\begin{frame}[t,shrink=4]{\sref{۳۴}{مطالعه حذفی: عمق و \cfg{}}}
\begin{noteslist}
\goal{نشان دهید انتخاب‌های طراحی از نظر هزینه و کیفیت بررسی شده‌اند.}
\say{بخش سمت راست جدول نشان می‌دهد افزایش عمق شاخه دوم از ۱ بلوک به ۲ و ۴ بلوک، هزینه محاسباتی، زمان هر دوره و حافظه را افزایش می‌دهد. بنابراین در نسخه نهایی، یک بلوک به عنوان نقطه تعادل استفاده شده است. بخش سمت چپ نشان می‌دهد استفاده از ویژگی‌های \vit{} کمی بهتر از اتصال ساده است. همچنین در این تنظیم حذفی، \cfg{} پایین‌تر نتیجه توزیعی بهتری داشته و با افزایش \cfg{}، \fid{} و \kid{} بدتر شده‌اند. با این حال، ارزیابی اصلی از \cfg{} برابر ۱٫۵ استفاده کرده تا تعادل بین وفاداری شرطی و تنوع حفظ شود.}
\defense{به داور بگویید ablationها نشان می‌دهند تصمیم‌ها فقط حدسی نبوده‌اند. البته بعضی ablationها محدود و کنترل‌شده‌اند، پس لحن باید محتاط باشد.}
\transition{حالا جمع‌بندی نتایج و محدودیت‌ها را مطرح می‌کنیم.}
\end{noteslist}
\end{frame}

\begin{frame}[t,shrink=4]{\sref{۳۵}{بحث و محدودیت‌ها}}
\begin{noteslist}
\goal{برداشت نهایی از نتایج را همراه با محدودیت‌ها بیان کنید.}
\say{نتایج نشان می‌دهد شاخه وضوح‌بالا می‌تواند به بهبود \kid{} و \lpips{} کمک کند. این موضوع با ایده تزریق ویژگی‌های محلی سازگار است. اما \fid{} نسبت به \lr{LightningDiT} بهتر نشده است؛ بنابراین روش پیشنهادی را باید مکمل بدانیم. همچنین مطالعه انسانی با \lpips{} هم‌سو است، ولی جایگزین تحلیل آماری چند اجرا نیست. محدودیت دیگر این است که شاخه محلی در برخی بافت‌های متراکم یا مرزهای مبهم ممکن است جزئیات نادرست را تقویت کند.}
\defense{این اسلاید برای اعتمادسازی است. اگر محدودیت‌ها را خودتان بگویید، پاسخ به نقد داور راحت‌تر می‌شود.}
\transition{بر اساس همین محدودیت‌ها، چند مسیر آینده پیشنهاد می‌شود.}
\end{noteslist}
\end{frame}

\begin{frame}[t,shrink=4]{\sref{۳۶}{پیشنهادات پژوهشی}}
\begin{noteslist}
\goal{آینده کار را به محدودیت‌های مشاهده‌شده وصل کنید.}
\say{اولین مسیر آینده فعال‌سازی \termfn{\lr{FiLM}}{۱} است. در نسخه فعلی، شاخه دوم می‌تواند از نظر طراحی با زمان و کلاس شرطی شود، اما این مسیر در آموزش اصلی فعال نشده است. برای دیتاست‌های پیچیده‌تر مثل \termfn{\lr{iNaturalist}}{۲} که تنوع ریزدانه و تفاوت‌های ظریف بین کلاس‌ها بیشتر است، شرط‌گذاری دقیق‌تر می‌تواند مهم باشد. مسیر دوم طراحی شاخه‌های چندمقیاسی است تا مدل فقط یک سطح محلی نبیند. مسیر سوم کاهش هزینه استنتاج با روش‌هایی مثل کوانتیزاسیون یا مسیرهای پویا است. همچنین می‌توان ایده را به تولید متن-به-تصویر، ویدئو، یا بدنه‌ها و خبره‌های جدیدتر تعمیم داد.}
\defense{پیشنهادها را به شکل فهرست آرزو نگویید؛ هر مورد را به یک محدودیت وصل کنید. مثلاً \lr{FiLM} برای شرط‌گذاری ریزدانه، چندمقیاسی برای جزئیات در مقیاس‌های مختلف، و کوانتیزاسیون برای هزینه استنتاج.}
\transition{در اسلاید بعدی جمع‌بندی نهایی پایان‌نامه را ارائه می‌کنم.}
\end{noteslist}
\notedefs{\footdef{۱}{مدولاسیون خطی ویژگی‌محور \lr{(Feature-wise Linear Modulation)}؛ شرطی‌سازی ویژگی‌ها با مقیاس و انتقال خطی وابسته به شرط.}
\footdef{۲}{مجموعه‌داده \lr{iNaturalist}: مجموعه‌داده زیستی با کلاس‌های ریزدانه و تنوع زیاد در گونه‌ها.}}
\end{frame}

\begin{frame}[t,shrink=4]{\sref{۳۷}{جمع‌بندی}}
\begin{noteslist}
\goal{پنج پیام اصلی پایان‌نامه را کوتاه و محکم تکرار کنید.}
\say{جمع‌بندی کار این است که \duo{} یک معماری دوجریانی برای تنظیم دقیق پارامتر-کارای \dit{} معرفی می‌کند. بدنه اصلی \lr{DiT-XL/2} منجمد می‌ماند و شاخه \lr{x2} ویژگی‌های محلی را استخراج می‌کند. مکانیزم توکن کلاس محلی، خروجی شاخه دوم را از \(4T\) به \(T\) توکن تبدیل می‌کند و این کار را به صورت محتوا-آگاه انجام می‌دهد. مدل با ۱۷٫۹۵ میلیون پارامتر آموزش‌پذیر به \lr{FID=2.219}، \lr{KID=0.0014} و \lr{LPIPS=0.7162} رسیده است. مطالعه انسانی هم نشان داده \duo{} در ۵۳٫۲۶ درصد مقایسه‌های غیرمساوی ترجیح داده شده است.}
\defense{این اسلاید را مثل پاسخ به سؤال «سهم پایان‌نامه چیست؟» ارائه کنید. اگر فقط یک جمله فرصت داشتید: «ما یک شاخه محلی و سبک برای تقویت \dit{} منجمد طراحی کردیم که با توکن کلاس محلی هم‌تراز می‌شود.»}
\transition{بعد از این اسلاید، وارد تشکر و پرسش‌ها می‌شویم.}
\end{noteslist}
\end{frame}

\begin{frame}[t,shrink=4]{\sref{۳۸}{سپاسگزارم / پرسش‌ها}}

\goal{ارائه را رسمی ببندید و برای پرسش آماده شوید.}

\defense{برای سؤال‌ها، سه محور را آماده داشته باشید: دلیل انتخاب شاخه \lr{x2}، تفاوت توکن کلاس محلی با pooling، و تفسیر منصفانه نتیجه نسبت به \lr{LightningDiT}.}
\transition{در صورت پرسش درباره جزئیات اجرا، از اسلاید پشتیبان تنظیمات آموزش استفاده کنید.}
\end{noteslist}

\begin{frame}[t,shrink=4]{\sref{۳۹}{پشتیبان: تنظیمات آموزش}}
\begin{noteslist}
\goal{در پاسخ به سؤال‌های فنی، جزئیات اجرایی را سریع و دقیق ارائه کنید.}
\say{این اسلاید برای بخش پرسش و پاسخ است. اگر داور درباره بازتولیدپذیری یا تنظیمات آموزش پرسید، از این جدول استفاده کنید. بگویید معماری پایه \lr{DiT-XL/2} بوده، اندازه پچ اصلی ۲ و شاخه دوم ۱ بوده، وضوح \(256\times256\) استفاده شده، \lr{VAE} همان \termfn{\lr{sd-vae-ft-ema}}{۱} بوده، فرآیند انتشار ۱۰۰۰ گام آموزشی داشته و نمونه‌برداری با ۲۵۰ گام \lr{DDPM} انجام شده است. بهینه‌ساز \lr{AdamW} و نرخ یادگیری \(10^{-4}\) استفاده شده و \cfg{} ارزیابی ۱٫۵ بوده است.}
\defense{اگر سؤال درباره سخت‌افزار یا چند seed مطرح شد و در اسلاید نیست، صادقانه بگویید در متن/گزارش فعلی آن بخش محدود است و به عنوان مسیر تکمیل ارزیابی قابل افزودن است.}
\transition{اگر پرسش درباره مبانی نظری یا منابع باشد، اسلاید پشتیبان منابع اصلی را نشان دهید.}

\notedefs{\footdef{۱}{وزن‌های \lr{sd-vae-ft-ema}: نسخه \lr{Stable Diffusion VAE} که با میانگین‌گیری نمایی وزن‌ها پایدار شده است.}}
